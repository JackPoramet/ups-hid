# UPS Monitor — Python dependencies
# Install with:  pip install -r requirements.txt

# ── Core HID ──────────────────────────────────────────────────────────────────
hidapi          # hid module — HID device communication (core_hid_ups.py)
pyusb           # usb.core / usb.util — USB access (libusb0 fallback)
libusb-package  # bundles libusb DLL for Windows

# ── GUI (Legacy desktop GUI) ──────────────────────────────────────────────────
PySide6         # Qt6 GUI framework (gui_windows.py, gui_linux.py)

# ── Web Server ────────────────────────────────────────────────────────────────
flask           # REST API + Web UI server (web_daemon.py, windows/tray_service/web_server.py)

# ── Windows Tray Service (windows/) ──────────────────────────────────────────
pystray         # System Tray Icon (windows/tray_service/tray_app.py)
Pillow          # สร้าง tray icon แบบ programmatic (pystray dependency)
plyer           # Windows Toast Notifications (windows/tray_service/notifications.py)
